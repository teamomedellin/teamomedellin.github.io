const SITIO = "https://teamomedellin.github.io";
const CATALOGO = SITIO + "/catalogo.json";
const LLAVES = { pruebas: "SKBHgdUSlIJaN3iABDkbDrCql94VoiEgTLKUZAR6a8o", produccion: "NTFQWI02n1FgvHKGIg8CLMLZ2S1bw7rgwYbZ2_nhLcs" };
const ENVIO = { tienda: 0, medellin: 15000, nacional: 18000 };
const TIENDAS = { calle10: "Calle 10 #37-41, El Poblado", manila: "Calle 13 #43D-27, Manila" };
const fmt = n => "$" + Math.round(n).toLocaleString("es-CO");

function cors(origin) {
  let ok = false;
  try { const h = new URL(origin).hostname; ok = origin === SITIO || /(^|\.)claudeusercontent\.com$|(^|\.)claude\.ai$/.test(h); } catch (_) {}
  return { "Access-Control-Allow-Origin": ok ? origin : SITIO, "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Vary": "Origin" };
}
const hex = buf => [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, "0")).join("");
async function sha256(t) { return hex(await crypto.subtle.digest("SHA-256", new TextEncoder().encode(t))); }
async function hmac(key, t) {
  const k = await crypto.subtle.importKey("raw", new TextEncoder().encode(key || "\0"), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  return hex(await crypto.subtle.sign("HMAC", k, new TextEncoder().encode(t)));
}
const clean = (s, n = 120) => String(s ?? "").replace(/[<>]/g, "").trim().slice(0, n);
const esc = s => String(s ?? "").replace(/[&<>"]/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));

async function crearPedido(body, env) {
  const { items, cliente = {}, entrega = {}, ref } = body;
  if (!Array.isArray(items) || !items.length || items.length > 60) throw new Error("pedido vacío");
  const tipo = entrega.tipo;
  if (!(tipo in ENVIO)) throw new Error("tipo de entrega no válido");
  const c = { nombre: clean(cliente.nombre, 80), tipoDoc: clean(cliente.tipoDoc, 4), doc: clean(cliente.doc, 20), cel: clean(cliente.cel, 20), correo: clean(cliente.correo, 80) };
  if (!c.nombre || !c.doc || !c.cel || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(c.correo)) throw new Error("faltan datos del cliente");
  const e = { tipo, tienda: clean(entrega.tienda, 10), dir: clean(entrega.dir, 120), detalle: clean(entrega.detalle, 80), barrio: clean(entrega.barrio, 60), ciudad: clean(entrega.ciudad, 60), depto: clean(entrega.depto, 60), notas: clean(entrega.notas, 200) };
  if (tipo === "tienda" && !(e.tienda in TIENDAS)) throw new Error("elige la tienda");
  if (tipo !== "tienda" && !e.dir) throw new Error("falta la dirección");
  if (tipo === "nacional" && (!e.ciudad || !e.depto)) throw new Error("falta ciudad o departamento");

  const cat = await (await fetch(CATALOGO, { cf: { cacheTtl: 300 } })).json();
  const opt = {};
  for (const p of cat.products) for (const o of p.options) opt[o.id] = { p, o };
  const porProd = {}, lineas = [];
  for (const it of items) {
    const x = opt[String(it.id)], q = Math.floor(Number(it.q));
    if (!x || !(q > 0) || q > 50) throw new Error("producto no válido");
    const d = Number.isInteger(it.d) && it.d >= 0 && it.d < x.p.photos.length ? it.d : (typeof x.o.ph === "number" ? x.o.ph : 0);
    lineas.push({ id: x.o.id, nombre: x.p.name, opcion: x.o.label || "", diseno: x.p.photos.length > 1 ? d + 1 : null, foto: x.p.photos[d] ? SITIO + "/" + x.p.photos[d] : "", q, precio: x.o.price });
    const k = x.p.key; (porProd[k] = porProd[k] || { p: x.p, l: [] }).l.push({ precio: x.o.price, q, max: x.o.q });
  }
  for (const { p, l } of Object.values(porProd)) {
    const qTot = l.reduce((a, b) => a + b.q, 0);
    if (l.some(b => b.q > b.max)) throw new Error("sin existencias suficientes: " + p.name);
  }
  let sub = 0;
  for (const { p, l } of Object.values(porProd)) {
    const qTot = l.reduce((a, b) => a + b.q, 0), base = l.reduce((a, b) => a + b.q * b.precio, 0);
    const m = String(p.promo || "").match(/(\d+)\s*x\s*\$?\s*([\d.]+)/i);
    sub += m ? Math.floor(qTot / +m[1]) * +m[2].replace(/\./g, "") + (qTot % +m[1]) * l[0].precio : base;
  }
  const envio = ENVIO[tipo], total = sub + envio;
  const orderId = "TAM-" + Date.now().toString(36).toUpperCase() + "-" + Math.random().toString(36).slice(2, 6).toUpperCase();
  const signature = await sha256(orderId + total + "COP" + env.BOLD_SECRET);
  const pedido = { orderId, creado: new Date().toISOString(), estado: "sin pagar", modo: env.MODO || "pruebas", cliente: c, entrega: e, lineas, subtotal: sub, envio, total, ref: clean(ref, 40) };
  if (env.PEDIDOS) await env.PEDIDOS.put("o:" + orderId, JSON.stringify(pedido), { expirationTtl: 60 * 60 * 24 * 400 });
  return { orderId, amount: total, subtotal: sub, envio, currency: "COP", signature };
}

async function marcarPagado(env, orderId, info) {
  if (!env.PEDIDOS) return null;
  const raw = await env.PEDIDOS.get("o:" + orderId); if (!raw) return null;
  const p = JSON.parse(raw);
  if (p.estado === "pagado" || p.estado === "despachado") return p;
  if (info.total && Number(info.total) !== p.total) { p.alerta = "El monto pagado (" + info.total + ") no coincide con el pedido"; }
  p.estado = "pagado"; p.pagado = new Date().toISOString(); p.bold = info;
  await env.PEDIDOS.put("o:" + orderId, JSON.stringify(p), { expirationTtl: 60 * 60 * 24 * 400 });
  p.aviso = await avisar(env, p);
  await env.PEDIDOS.put("o:" + orderId, JSON.stringify(p), { expirationTtl: 60 * 60 * 24 * 400 });
  return p;
}

async function avisar(env, p) {
  const ent = p.entrega.tipo === "tienda" ? "Recoge en " + (p.entrega.tienda === "manila" ? "Manila" : "Calle 10") : p.entrega.tipo === "medellin" ? "Domicilio Medellín · " + p.entrega.barrio : "Envío nacional · " + p.entrega.ciudad;
  const n = p.lineas.reduce((a, l) => a + l.q, 0);
  return enviarAviso(env, "Nuevo pedido pagado " + fmt(p.total) + (p.modo === "pruebas" ? " (PRUEBA)" : ""), p.orderId + " · " + n + (n === 1 ? " producto" : " productos") + "\n" + ent + "\n" + p.cliente.nombre);
}

async function enviarAviso(env, titulo, texto) {
  const panelUrl = "https://" + env.WORKER_HOST + "/pedidos";
  try {
    if (env.TELEGRAM_TOKEN) {
      if (!env.TELEGRAM_CHAT) return "falta la variable TELEGRAM_CHAT";
      const r = await fetch("https://api.telegram.org/bot" + String(env.TELEGRAM_TOKEN).trim() + "/sendMessage", {
        method: "POST", headers: { "content-type": "application/json" },
        body: JSON.stringify({ chat_id: String(env.TELEGRAM_CHAT).trim(), text: "🛍️ " + titulo + "\n\n" + texto, reply_markup: { inline_keyboard: [[{ text: "Ver pedidos", url: panelUrl }]] } }),
      });
      return r.ok ? "ok" : "Telegram respondió " + r.status + ": " + (await r.text()).slice(0, 200);
    }
    if (!env.NTFY_TOPIC) return "faltan las variables de Telegram";
    const r = await fetch("https://ntfy.sh/", {
      method: "POST",
      headers: { "content-type": "application/json", ...(env.NTFY_TOKEN ? { Authorization: "Bearer " + String(env.NTFY_TOKEN).trim() } : {}) },
      body: JSON.stringify({ topic: String(env.NTFY_TOPIC).trim(), title: titulo, message: texto, tags: ["shopping_bags", "moneybag"], priority: 4, click: panelUrl }),
    });
    return r.ok ? "ok" : "ntfy respondió " + r.status + ": " + (await r.text()).slice(0, 200);
  } catch (e) { return "error: " + String(e.message || e); }
}

async function consultarBold(env, orderId) {
  const key = LLAVES[env.MODO === "produccion" ? "produccion" : "pruebas"];
  const r = await fetch("https://payments.api.bold.co/v2/payment-voucher/" + encodeURIComponent(orderId), { headers: { Authorization: "x-api-key " + key } });
  if (!r.ok) return { payment_status: "ERROR_" + r.status };
  return r.json();
}

async function listar(env) {
  const out = []; let cursor;
  do { const r = await env.PEDIDOS.list({ prefix: "o:", cursor, limit: 1000 }); for (const k of r.keys) out.push(k.name); cursor = r.list_complete ? null : r.cursor; } while (cursor && out.length < 2000);
  const vals = await Promise.all(out.slice(-200).map(k => env.PEDIDOS.get(k)));
  return vals.filter(Boolean).map(v => JSON.parse(v)).sort((a, b) => (b.pagado || b.creado).localeCompare(a.pagado || a.creado));
}

  async fetch(req, env) {
    const url = new URL(req.url); env.WORKER_HOST = url.host;
    const h = cors(req.headers.get("Origin"));
    if (req.method === "OPTIONS") return new Response(null, { headers: h });
    try {
      if (url.pathname === "/webhook" && req.method === "POST") {
        const body = await req.text();
        const firma = await hmac(env.MODO === "produccion" ? env.BOLD_SECRET : "", btoa(unescape(encodeURIComponent(body))));
        if (firma !== (req.headers.get("x-bold-signature") || "")) return new Response("firma inválida", { status: 401 });
        const ev = JSON.parse(body), ref = ev?.data?.metadata?.reference;
        if (ev.type === "SALE_APPROVED" && ref) await marcarPagado(env, ref, { transaction_id: ev.subject, total: ev?.data?.amount?.total, via: "webhook" });
        return new Response("ok");
      }
      if (url.pathname === "/confirmar" && req.method === "POST") {
        const { orderId } = await req.json();
        const v = await consultarBold(env, clean(orderId, 60));
        let p = null;
        if (v.payment_status === "APPROVED") p = await marcarPagado(env, orderId, { transaction_id: v.transaction_id, total: v.total, metodo: v.payment_method, via: "consulta" });
        return Response.json({ estado: v.payment_status, total: p ? p.total : v.total }, { headers: h });
      }
      if (url.pathname === "/telegram-id") {

        if (!env.TELEGRAM_TOKEN) return new Response("Primero guarda TELEGRAM_TOKEN en Cloudflare y dale Deploy", { headers: { "content-type": "text/plain; charset=utf-8" } });

        if (req.method === "POST") { const f = await req.formData(); pin = f.get("pin") || ""; 
          if (pin && pin === env.PANEL_PIN && env.PEDIDOS) { const id = f.get("id"); const raw = await env.PEDIDOS.get("o:" + id); if (raw) { const p = JSON.parse(raw); p.estado = "despachado"; p.despachado = new Date().toISOString(); await env.PEDIDOS.put("o:" + id, JSON.stringify(p), { expirationTtl: 60 * 60 * 24 * 400 }); } } }
